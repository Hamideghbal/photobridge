 const norm=x=>String(x??'').trim().toUpperCase();
 function returnMatch(slots,raw){
  const text=String(raw??'').trim();if(!text||text.length>512)return {error:'invalid'};
  const qr=/^PB\|/i.test(text),parts=text.split('|'),value=qr?parts[1]:text;
  if(!value)return {error:'invalid'};
  const k=norm(value),own=slots.filter(s=>!s.isExternal&&s.code);
  let found=own.filter(s=>norm(s.code)===k);
  if(!found.length&&/^PB-/i.test(k))found=own.filter(s=>norm(s.code)===k.slice(3));
  if(!found.length&&!qr)found=own.filter(s=>s.serial&&norm(s.serial)===k);
  if(found.length!==1)return {error:found.length?'ambiguous':'not_order'};
  return found[0].checked?{error:'duplicate'}:{slot:found[0]};
 }
 function queue(onCode,onPending,onStatus){
  let pending=0,tail=Promise.resolve(),closed=false;
  return {get pending(){return pending},close(){closed=true},push(raw){
   raw=String(raw??'').trim();if(closed||!raw||raw.length>512)return Promise.resolve(false);
   if(pending>=32){onStatus('صف اسکن پر است؛ پس از بررسی کدها دوباره اسکن کنید.',true);return Promise.resolve(false)}
   pending++;onPending(pending);
   const next=tail.then(async()=>{if(closed)return false;try{return await onCode(raw)}catch{if(!closed)onStatus('بررسی کد انجام نشد؛ دوباره اسکن کنید.',true);return false}});
   tail=next.finally(()=>{pending--;if(!closed)onPending(pending)});
   return tail.then(()=>next);
  }};
 }
