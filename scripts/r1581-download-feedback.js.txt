/* R1581: native notification progress and invoice preview use the existing verified download pipeline. */
const pb1581Transfers=new Map(),pb1581Invoices=[],pb1581Shown=new Set();
function pb1581Report(order,value){
 const reason=value.phase==='failed'?pb1582Explain(order,value.error,value.explain):'';if(value.phase==='verified'){for(const key of pb1582Problems.keys())if(key.startsWith((order?.lab_id||'')+':'))pb1582Problems.delete(key);}
 if(!window.pbNativeStorage||typeof window.pbDesktopEvents?.orderProgress!=='function'||!order?.id||!order.lab_id)return;
 const key=order.lab_id+':'+order.id,old=pb1581Transfers.get(key)||{last:0},phase=value.phase==='failed'?'failed':value.phase==='verified'?'verified':value.phase==='downloading'?'downloading':'checking';
 old.value={...old.value,...value,labId:order.lab_id,orderId:order.id,number:order.number,phase,reason};
 const send=()=>{old.timer=null;old.last=Date.now();void window.pbDesktopEvents.orderProgress(old.value).catch(()=>{})};
 clearTimeout(old.timer);if(phase==='verified'||phase==='failed'||Date.now()-old.last>=250)send();else old.timer=setTimeout(send,250-(Date.now()-old.last));
 pb1581Transfers.set(key,old);if(pb1581Transfers.size>50){const first=pb1581Transfers.keys().next().value;clearTimeout(pb1581Transfers.get(first).timer);pb1581Transfers.delete(first)}
}
function pb1581OpenInvoice(invoice,orderId,labId){
 if(!window.pbNativeStorage||!invoice?.id||!labId)return;
 const key=labId+':'+invoice.id;if(pb1581Shown.has(key))return;pb1581Shown.add(key);
 pb1581Invoices.push({labId,orderId,invoice});
 if(!/^\/lab\/(orders|album)(?:\/|$)/.test(location.pathname))Dn('/lab/orders/flow');
 void window.pbDesktopEvents?.focusForPricing?.(labId)?.catch(()=>{});
 dispatchEvent(new CustomEvent('photobridge-download-invoice'));
}
addEventListener('photobridge-auth-session-changed',()=>{for(const item of pb1581Transfers.values())clearTimeout(item.timer);pb1581Transfers.clear();pb1581Invoices.length=0;pb1581Shown.clear()});
