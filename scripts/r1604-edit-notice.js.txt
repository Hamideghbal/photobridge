function pb1604EditNotice(message){
 const dialog=document.createElement('dialog');dialog.className='pb1604EditNotice';dialog.dir='rtl';
 const heading=document.createElement('h3');heading.textContent='ویرایش عکس ذخیره شد';
 const body=document.createElement('p');body.textContent=message;
 const button=document.createElement('button');button.type='button';button.className='primary';button.textContent='بازگشت به عکس‌ها';
 const previous=document.activeElement;button.onclick=()=>dialog.close();dialog.addEventListener('close',()=>{dialog.remove();if(previous?.isConnected)previous.focus()},{once:true});dialog.append(heading,body,button);document.body.append(dialog);dialog.showModal();button.focus();
}
