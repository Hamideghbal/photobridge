  function pb2040WatchLifecycle(){
    if(!mobile()||!document.body)return;
    // Observe the stable document body, not a detached React workspace.
    // Schedule only a genuine entry, exit or missing control; no idle writes.
    const observer=new MutationObserver(()=>{
      const w=$('.photographerMain.photographerPrintWorkspace'),row=$('#pbF37PrintComposerActions');
      if(pb2036MobilePhoto(w)?pb2040NeedsPhotoRepair(w,row):!!pb2036PhotoWorkspace)schedule();
    });
    observer.observe(document.body,{subtree:true,childList:true,attributes:true,attributeFilter:['class','hidden']});
  }

  function pb2036SyncPhoto(row,w){
    if(!pb2036MobilePhoto(w)||!row?.isConnected)return;
    if(!document.documentElement.classList.contains('pb2036PhotoComposer'))document.documentElement.classList.add('pb2036PhotoComposer');
    if(row.hidden)row.hidden=false;
    if(!row.classList.contains('pb2036PhotoActions'))row.classList.add('pb2036PhotoActions');
    const choose=row.querySelector('.pbF37Choose'),send=row.querySelector('.pbF37Send');
    let product=row.querySelector('.pb2036Standalone');
    if(!product){
      product=document.createElement('button');product.type='button';product.className='pb2036Standalone';
      product.textContent='سفارش شاسی و قاب بدون چاپ';
      product.addEventListener('click',()=>{const source=pb2036ProductSource($('.photographerMain.photographerPrintWorkspace'));if(source&&!source.disabled)source.click()});
      row.append(product);
    }
    const sources=[pb2040PhotoAction(w,0),pb2040PhotoAction(w,1),pb2036ProductSource(w)];
    for(const [index,button] of [choose,send,product].entries()){
      const disabled=!sources[index]||!!sources[index].disabled;
      if(button&&button.disabled!==disabled)button.disabled=disabled;
    }
    if(choose&&choose.textContent!=='انتخاب سایز و عکس')choose.textContent='انتخاب سایز و عکس';
    if(send&&send.textContent!=='ارسال')send.textContent='ارسال';
    if(send&&send.getAttribute('aria-label')!=='ارسال فایل برای چاپ')send.setAttribute('aria-label','ارسال فایل برای چاپ');
  }

  }

    const row=$('#pbF37PrintComposerActions');if(row){row.classList.remove('pb2036PhotoActions');row.querySelector('.pb2036Standalone')?.remove();row.querySelector('.pbF37Send')?.removeAttribute('aria-label')}
  }
