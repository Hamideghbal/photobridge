/* R1584: stage percentage tariff edits in the existing form; never save or issue invoices here. */
function pb1584Digits(value){return String(value??'').replace(/[۰-۹]/g,c=>'۰۱۲۳۴۵۶۷۸۹'.indexOf(c)).replace(/[٠-٩]/g,c=>'٠١٢٣٤٥٦٧٨٩'.indexOf(c)).replace(/٫/g,'.').trim()}
function pb1584PlanIncrease(rows,{fields,percent,mode='none',step='1000'}){
 const p=pb1584Digits(percent);if(!/^\d+(?:\.\d{1,2})?$/.test(p)||p.length>14)throw Error('درصد افزایش را با حداکثر دو رقم اعشار وارد کنید.');
 const [whole,fraction='']=p.split('.'),basis=BigInt(whole)*100n+BigInt(fraction.padEnd(2,'0'));if(basis<=0n)throw Error('درصد افزایش باید بیشتر از صفر باشد.');
 if(!['none','up','down'].includes(mode))throw Error('روش گردکردن معتبر نیست.');
 const scale=pb1584Digits(step).replace(/[٬,]/g,'');if(mode!=='none'&&(!/^\d+$/.test(scale)||BigInt(scale)<=0n||BigInt(scale)>BigInt(Number.MAX_SAFE_INTEGER)))throw Error('گام گردکردن باید مبلغی صحیح و بیشتر از صفر، به تومان باشد.');
 const selected=[...new Set(fields||[])];if(!selected.length)throw Error('حداقل یک بخش تعرفه را انتخاب کنید.');
 if(selected.some(key=>!['luster','silk','metallic','chassis','frame','album_binding_price'].includes(key)&&!/^pb1539_[A-Za-z0-9_-]+$/.test(key)))throw Error('انتخاب ستون تعرفه معتبر نیست.');
 const entries=[],changes=[];let empty=0,protectedCount=0;
 const next=rows.map(row=>{const copy={...row};for(const key of selected){const raw=row[key];if(raw==null||String(raw).trim()===''){empty++;continue;}const text=pb1584Digits(raw).replace(/[٬,]/g,'');if(!/^\d+$/.test(text)||BigInt(text)>BigInt(Number.MAX_SAFE_INTEGER))throw Error('قیمت «'+(row.label||row.id)+'» باید مبلغ صحیح و نامنفی به تومان باشد.');const old=BigInt(text),numerator=old*(10000n+basis);let value;
  if(mode==='none')value=(numerator+5000n)/10000n;else{const unit=BigInt(scale),denominator=10000n*unit;value=(mode==='up'?(numerator+denominator-1n)/denominator:numerator/denominator)*unit;}
  let protectedValue=false;if(value<old){value=old;protectedCount++;protectedValue=true;}if(value>BigInt(Number.MAX_SAFE_INTEGER))throw Error('قیمت نهایی از محدودهٔ مجاز بیشتر می‌شود؛ درصد یا گام گردکردن را کمتر کنید.');
  const entry={id:row.id,label:row.label||row.id,field:key,before:Number(old),after:Number(value),protected:protectedValue};entries.push(entry);if(value!==old){copy[key]=Number(value);changes.push(entry);}
 }return copy;});return{rows:next,entries,changes,empty,protectedCount};
}
function pb1584OpenPriceIncrease({title,getRows,groups,onApply,fieldLabels={}}){
 const existing=document.querySelector('dialog.pb1584Markup');if(existing){existing.focus();return;}
 const previous=document.activeElement,owner=document.querySelector('.pricingSettingsCard'),dialog=document.createElement('dialog');dialog.className='pb1584Markup';dialog.dir='rtl';dialog.setAttribute('aria-label','افزایش درصدی '+title);
 const node=(tag,props={},text)=>{const element=document.createElement(tag);Object.assign(element,props);if(text!=null)element.textContent=text;return element};
 const header=node('header'),heading=node('h2',{},'افزایش درصدی '+title),close=node('button',{type:'button',className:'pb1584Close'},'بستن');close.onclick=()=>dialog.close();header.append(heading,close);
 const form=node('form'),help=node('p',{className:'pb1584Help'},'فقط تعرفه‌های همین جدول تغییر می‌کنند؛ موجودی‌ها و فاکتورهای قبلی تغییر نمی‌کنند. قیمت‌ها به تومان صحیح ذخیره می‌شوند.');
 const choices=node('fieldset'),legend=node('legend',{},'بخش‌های موردنظر');choices.append(legend);const selected=[];
 for(const group of groups){if(!group.fields.length)continue;const label=node('label'),input=node('input',{type:'checkbox'});input.setAttribute('aria-label',group.label);selected.push({input,fields:group.fields});label.append(input,document.createTextNode(group.label));choices.append(label);}
 const controls=node('div',{className:'pb1584Controls'}),percent=node('input',{type:'text',inputMode:'decimal',placeholder:'مثلاً ۱۰ یا ۱۲٫۵'}),mode=node('select'),step=node('input',{type:'text',inputMode:'numeric',value:'1000',disabled:true});percent.setAttribute('aria-label','درصد افزایش');step.setAttribute('aria-label','گام گردکردن به تومان');mode.setAttribute('aria-label','روش گردکردن');
 for(const [value,label]of [['none','بدون گردکردن اضافه'],['up','گردکردن رو به بالا'],['down','گردکردن رو به پایین']])mode.append(node('option',{value},label));
 for(const [label,input]of [['درصد افزایش',percent],['روش گردکردن',mode],['گام گردکردن (تومان)',step]]){const wrapper=node('label',{},label);wrapper.append(input);controls.append(wrapper)}
 const minimumLabel=node('p',{className:'pb1584Help'},'اگر گردکردن رو به پایین قیمت را کمتر کند، قیمت قبلی حفظ می‌شود.');
 const hint=node('p',{className:'pb1584Help'},'مثلاً با گام ۱۰۰۰، مبلغ ۱۲٬۳۴۵ رو به بالا ۱۳٬۰۰۰ و رو به پایین ۱۲٬۰۰۰ می‌شود.'),previewButton=node('button',{type:'submit',className:'pb1584Preview'},'پیش‌نمایش قیمت‌ها'),status=node('p',{className:'pb1584Status'});status.setAttribute('role','status');const preview=node('div',{className:'pb1584Table'}),footer=node('footer'),note=node('p',{className:'pb1584Help'},'پس از انتقال قیمت‌ها به جدول، برای ثبت نهایی دکمهٔ «ذخیره تعرفه» را بزنید.'),apply=node('button',{type:'button',className:'pb1584Apply',disabled:true},'انتقال قیمت‌ها به جدول');footer.append(note,apply);
 let plan=null,snapshot='',applied=false;const invalidate=()=>{plan=null;apply.disabled=true;preview.replaceChildren();status.textContent=''};form.addEventListener('input',invalidate);form.addEventListener('change',()=>{step.disabled=mode.value==='none';invalidate()});
 form.onsubmit=event=>{event.preventDefault();invalidate();try{const rows=getRows();snapshot=JSON.stringify(rows);plan=pb1584PlanIncrease(rows,{fields:selected.filter(v=>v.input.checked).flatMap(v=>v.fields),percent:percent.value,mode:mode.value,step:step.value});status.textContent=plan.changes.length.toLocaleString('fa-IR')+' قیمت آمادهٔ تغییر؛ '+plan.empty.toLocaleString('fa-IR')+' تعرفهٔ خالی بدون تغییر'+(plan.protectedCount?'؛ '+plan.protectedCount.toLocaleString('fa-IR')+' قیمت برای جلوگیری از کاهش ثابت ماند.':'');
 const table=node('table'),head=node('thead'),tr=node('tr');for(const text of ['سایز / بخش','قیمت قبلی','قیمت جدید','تغییر'])tr.append(node('th',{},text));head.append(tr);table.append(head);const body=node('tbody'),labels=Object.fromEntries(groups.flatMap(g=>g.fields.map(field=>[field,g.label])));
 for(const item of plan.entries.slice(0,200)){const row=node('tr',{className:item.after<item.before?'decrease':''});for(const text of [item.label+' / '+(fieldLabels[item.field]||labels[item.field]),item.before.toLocaleString('fa-IR'),item.after.toLocaleString('fa-IR'),(item.after-item.before).toLocaleString('fa-IR')])row.append(node('td',{},text));body.append(row)}table.append(body);preview.append(table);if(plan.entries.length>200)preview.append(node('p',{},'نمایش ۲۰۰ مورد اول؛ تغییر روی همهٔ ردیف‌های انتخاب‌شده اعمال می‌شود.'));apply.disabled=!plan.changes.length;
 }catch(error){status.textContent=error.message;plan=null;apply.disabled=true;}};
 apply.onclick=()=>{if(!plan||applied)return;if(owner&&!owner.isConnected){invalidate();status.textContent='صفحهٔ تعرفه تغییر کرده است؛ پنجره را ببندید و دوباره از صفحهٔ تعرفه باز کنید.';return;}if(JSON.stringify(getRows())!==snapshot){invalidate();status.textContent='قیمت‌ها بعد از پیش‌نمایش تغییر کرده‌اند؛ دوباره پیش‌نمایش بگیرید.';return;}applied=true;apply.disabled=true;try{onApply(plan.rows);dialog.close();}catch(error){applied=false;status.textContent=error.message;apply.disabled=false;}};
 form.append(help,choices,controls,minimumLabel,hint,previewButton,status,preview);dialog.append(header,form,footer);dialog.addEventListener('close',()=>{dialog.remove();if(previous?.isConnected)previous.focus()},{once:true});document.body.append(dialog);dialog.showModal();percent.focus();
}
