// R1714: finalize print uploads with bounded polling; retry lost gateway acknowledgements.
async function pb1714AwaitUploadReady(request, fileId, signal, originalFile) {
    let response;
    try {
        response = await request('/api/customer/uploads/complete', { method: 'POST', body: JSON.stringify({ fileId }), signal });
    } catch (error) {
        // A 502/504 after the PUT does not prove the uploaded original is missing.
        if (![502, 503, 504].includes(Number(error?.status))) throw error;
        response = { processing: true };
    }
    if (!response?.processing) return response;
    let deadline = Date.now() + 240000;
    let directRecoveryAttempted = false;
    let jpegReconfirmAttempts = 0;
    const reconfirm = () => request('/api/customer/uploads/complete', {
        method: 'POST', body: JSON.stringify({ fileId }), signal
    });
    while (Date.now() < deadline) {
        if (signal?.aborted) {
            const error = new Error('upload_aborted');
            error.name = 'AbortError';
            throw error;
        }
        await new Promise(resolve => setTimeout(resolve, 1600));
        try {
            const status = await request('/api/customer/uploads/status/' + encodeURIComponent(fileId), { method: 'GET', signal });
            if (status?.ok && !status.processing) return status;
        } catch (error) {
            if (!directRecoveryAttempted && originalFile &&
                Number(error?.status) === 422 &&
                error?.data?.phase === 'read' &&
                error?.data?.recoveryAvailable) {
                directRecoveryAttempted = true;
                // Original file still lives in this browser. Retry directly to
                // the authenticated app only after the S3 GET read has failed.
                try {
                    const fallback = await request('/api/customer/uploads/' + encodeURIComponent(fileId) + '/direct-heic', {
                        method: 'PUT', body: originalFile, signal,
                        headers: { 'content-type': 'application/octet-stream' }
                    });
                    if (fallback?.ok && !fallback?.processing) return fallback;
                } catch (fallbackError) {
                    // A 502 after this PUT is ambiguous; poll the SAME fileId.
                    if (![502, 503, 504].includes(Number(fallbackError?.status))) throw fallbackError;
                }
                deadline = Date.now() + 240000;
                continue;
            }
            if (Number(error?.status) === 409 &&
                error?.data?.error === 'upload_not_confirmed' &&
                jpegReconfirmAttempts < 2) {
                // A gateway 502 during ordinary JPEG confirmation is ambiguous.
                // Safely replay the same fileId; server completion is idempotent.
                jpegReconfirmAttempts++;
                try {
                    const renewed = await reconfirm();
                    if (renewed?.ok && !renewed?.processing) return renewed;
                } catch (e) {
                    if (![502, 503, 504].includes(Number(e?.status))) throw e;
                }
                continue;
            }
            if ([502, 503, 504].includes(Number(error?.status))) continue;
            throw error;
        }
    }
    throw Object.assign(new Error('print_jpeg_conversion_timeout'), {
        data: { error: 'print_jpeg_conversion_failed', reason: 'تبدیل عکس بیش از حد طول کشید. فایل اصلی حفظ شده است و می‌توان دوباره تلاش کرد.' }
    });
}
