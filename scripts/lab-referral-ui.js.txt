function PBLabWalletSubscription({plan,balance,onSaved,labId}){
 const[open,setOpen]=x.useState(false),[busy,setBusy]=x.useState(false),[error,setError]=x.useState(''),[done,setDone]=x.useState(null),intent=x.useRef(null);if(!plan||Number(balance||0)<=0)return null;
 const storageKey='lab-subscription-wallet-intent:'+labId+':'+plan.id;
 const restore=()=>{if(intent.current)return intent.current;try{const saved=JSON.parse(sessionStorage.getItem(storageKey)||'null');if(saved?.key&&Number.isSafeInteger(saved.amount))intent.current=saved}catch{}return intent.current};
 const clear=()=>{intent.current=null;try{sessionStorage.removeItem(storageKey)}catch{}};
 const pay=async()=>{if(busy)return;setBusy(true);setError('');let request=restore();if(!request){request={key:crypto.randomUUID(),amount:Number(plan.price)};intent.current=request;try{sessionStorage.setItem(storageKey,JSON.stringify(request))}catch{}}
 try{const result=await Me(window.pbNativeStorage?'/api/lab/subscription/zibal/request':'/api/lab/subscription/wallet',{method:'POST',body:JSON.stringify({paymentMethod:'wallet',planId:plan.id,idempotencyKey:request.key,expectedAmount:request.amount})});if(result?.renewalId!==request.key||Number(result.amount)!==request.amount||!Number.isFinite(Date.parse(result.activeUntil)))throw Error('payment_not_confirmed');setDone(result);setOpen(false);clear();try{await onSaved?.()}catch{setError('اشتراک تمدید شد؛ تازه‌سازی صفحه انجام نشد.')}}catch(err){const code=err?.data?.error;if(code==='subscription_price_changed'){clear();setError('قیمت طرح تغییر کرده است؛ مبلغ تازه را بررسی و دوباره تأیید کنید.');try{await onSaved?.()}catch{}}else setError(code==='wallet_insufficient'?'اعتبار کیف پول کافی نیست.':'نتیجهٔ درخواست هنوز تأیید نشده است؛ بررسی دوباره با همان شناسه انجام می‌شود تا برداشت تکراری نشود.')}finally{setBusy(false)}};
 return e.jsxs('div',{className:'labReferralWalletCheckout',children:[e.jsx('button',{type:'button',className:'pbActionButton',onClick:()=>{restore();if(!intent.current){setError('');setDone(null)}setOpen(true)},children:'تمدید با کیف پول · '+V(balance)+' تومان'}),done&&e.jsx('p',{role:'status',children:'اشتراک با کیف پول تمدید شد.'}),!open&&error&&e.jsx('p',{role:'status',children:error}),open&&e.jsx(On,{title:'پرداخت اشتراک از کیف پول',onClose:()=>{if(!busy)setOpen(false)},children:e.jsxs('div',{className:'labReferralWalletConfirm',children:[e.jsx('p',{children:'طرح: '+plan.name}),e.jsx('p',{children:'مبلغ درخواست: '+V(intent.current?.amount??plan.price)+' تومان'}),e.jsx('p',{children:intent.current?'این درخواست قبلاً ارسال شده است. نتیجه با همان شناسه بررسی می‌شود و برداشت دوباره انجام نمی‌شود.':'با تأیید شما، همین مبلغ یک‌بار از اعتبار کیف پول کم می‌شود و اشتراک تمدید می‌شود.'}),error&&e.jsx('p',{role:'alert',children:error}),e.jsxs('div',{className:'modalActions',children:[e.jsx('button',{type:'button',disabled:busy,onClick:()=>setOpen(false),children:'انصراف'}),e.jsx('button',{type:'button',className:'primary',disabled:busy,onClick:pay,children:busy?'در حال بررسی پرداخت…':intent.current?'بررسی و ادامهٔ درخواست قبلی':'تأیید و پرداخت از کیف پول'})]})]})})]});
}
