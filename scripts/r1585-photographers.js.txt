/* R1585: display the existing finance discount settings; no discount or membership rules change. */
function pb1585HasLocation(customer){return customer.location_lat!=null&&customer.location_lng!=null&&String(customer.location_lat).trim()!==''&&String(customer.location_lng).trim()!==''&&Ra(customer.location_lat,customer.location_lng)}
function pb1585DiscountInfo(customer){
 if(!Object.hasOwn(customer,'finance_discount_percent'))return null;
 const percent=Number(customer.finance_discount_percent),rawEvery=Number(customer.finance_discount_every_invoices);
 if(!Number.isFinite(percent)||percent<0||percent>100)return null;
 return{percent,enabled:customer.finance_discount_enabled===true,configured:customer.finance_discount_configured===true,every:Number.isInteger(rawEvery)&&rawEvery>0?rawEvery:10};
}
function PB1585Discount({customer}){const info=pb1585DiscountInfo(customer);return e.jsxs('div',{className:'pb1585Discount',children:[e.jsx('strong',{children:info?V(info.percent)+'٪':'—'}),e.jsx('small',{children:!info?'هنوز دریافت نشده':!info.configured?'تعریف نشده':info.enabled?'هر '+V(info.every)+' فاکتور':'غیرفعال'})]})}
function PB1585CopyLocation({customer}){return e.jsx('button',{type:'button',className:'pb1585CopyCoordinates',onClick:async()=>{const value=Number(customer.location_lat).toFixed(7)+', '+Number(customer.location_lng).toFixed(7);try{await navigator.clipboard.writeText(value);alert('مختصات عکاسی کپی شد.')}catch{prompt('مختصات را کپی کن:',value)}},children:'کپی مختصات'})}

 ]},customer.id))})]})});
