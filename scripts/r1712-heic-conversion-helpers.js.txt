// R1713: retry print-original downloads with Liara-compatible path-style GET.
async function pb1712ReadPrintOriginal(row, target) {
    const key = String(row?.object_key || '');
    if (!key) throw new Error('print_source_key_missing');
    // R1716: when S3 HEAD/PUT succeeds but GET cannot retrieve the original,
    // the authenticated uploader may stream the same HEIC bytes directly to
    // this API instance. The verified file is scoped to this order file id.
    const direct = typeof pb1716DirectSources !== 'undefined'
        ? pb1716DirectSources.get(String(row?.id || '')) : null;
    if (direct) {
        const stat = await fs.promises.stat(direct.path);
        const expected = Number(row?.bytes || 0);
        if (!stat.isFile() || !stat.size || (expected > 0 && stat.size !== expected))
            throw new Error('print_direct_source_size_mismatch');
        return direct.path;
    }
    let last;
    for (let attempt = 0; attempt < 3; attempt++) {
        const pathStyle = attempt > 0 && !key.startsWith('local/') && !!s3Download;
        const signal = AbortSignal.timeout(pb1666IsHeic(row) ? 20000 : 60000);
        try {
            let src;
            if (pathStyle) {
                const object = await s3Download.send(new GetObjectCommand({ Bucket: S3_BUCKET, Key: key }), { abortSignal: signal });
                if (!object?.Body) throw new Error('print_source_body_missing');
                await fs.promises.mkdir(path.dirname(target), { recursive: true });
                if (typeof object.Body.pipe === 'function') {
                    await streamPipeline(object.Body, fs.createWriteStream(target), { signal });
                } else if (typeof object.Body.transformToByteArray === 'function') {
                    await fs.promises.writeFile(target, Buffer.from(await object.Body.transformToByteArray()));
                } else {
                    const chunks = [];
                    for await (const chunk of object.Body) chunks.push(Buffer.isBuffer(chunk) ? chunk : Buffer.from(chunk));
                    await fs.promises.writeFile(target, Buffer.concat(chunks));
                }
                src = target;
            } else {
                src = await materializeAiObjectForComposite(key, target, signal);
            }
            const stat = await fs.promises.stat(src);
            const expected = Number(row?.bytes || 0);
            if (!stat.isFile() || !stat.size || (expected > 0 && stat.size !== expected))
                throw new Error('print_source_size_mismatch');
            return src;
        } catch (error) {
            last = error;
            console.warn('[print-jpeg:source-read-retry]', {
                fileId: row?.id,
                attempt: attempt + 1,
                mode: pathStyle ? 'path-style' : 'default',
                code: String(error?.name || error?.code || error?.Code || 'unavailable').slice(0, 90),
                httpStatus: Number(error?.$metadata?.httpStatusCode) || undefined
            });
            if (!key.startsWith('local/')) await fs.promises.rm(target, { force: true }).catch(() => {});
            if (attempt < 2) await new Promise(resolve => setTimeout(resolve, 250 * (attempt + 1)));
        }
    }
    throw last || new Error('print_source_unavailable');
}

// R1712: recognize HEIC by its file signature, not only its name.
async function pb1712LooksHeic(file){
 const handle=await fs.promises.open(file,'r');
 try{
  const buf=Buffer.alloc(96),{bytesRead}=await handle.read(buf,0,buf.length,0);
  if(bytesRead<16||buf.toString('ascii',4,8)!=='ftyp')return false;
  const brands=buf.toString('ascii',8,Math.min(bytesRead,96));
  return /heic|heix|heim|heis|hevc|hevx|mif1|msf1/i.test(brands);
 }finally{await handle.close()}
}
async function pb1712ValidJpeg(file){
 try{
  const stat=await fs.promises.stat(file);
  if(stat.size<100)return false;
  const meta=await sharp(file,{failOn:'error',pages:1}).metadata();
  return meta.format==='jpeg'&&meta.width>0&&meta.height>0;
 }catch{return false}
}

function pb1712FailureReason(phase,message){
 if(phase==='storage')return 'فایل JPEG آماده شد، اما ذخیره نسخه چاپ در سرور ناموفق بود؛ دوباره تلاش کنید';
 if(phase==='database')return 'فایل JPEG آماده شد، اما ثبت نتیجه در سفارش ناموفق بود؛ دوباره تلاش کنید';
 if(phase==='read')return 'فایل اصلی برای تبدیل از فضای ذخیره‌سازی دریافت نشد';
 if(phase==='verify')return 'خروجی تبدیل، JPEG معتبر و قابل چاپ نبود';
 return printJpegFailureReason(message);
}
