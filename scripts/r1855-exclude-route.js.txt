        // Explicit removal from a print draft only; preserve original storage bytes.
        const pb1855Exclude=url.pathname.match(/^\/api\/customer\/uploads\/([0-9a-f-]+)\/exclude$/);
        if(pb1855Exclude&&req.method==='POST'){
            const body=await bodyJson(req),orderId=String(body.orderId||'');
            if(!pb1854Uuid(orderId))return sendJson(res,400,{error:'invalid_order'});
            const client=await pool.connect();
            try{
                await client.query('BEGIN');
                const order=await client.query("SELECT status FROM orders WHERE id=$1 AND lab_id=$2 AND customer_id=$3 AND mode='print' FOR UPDATE",[orderId,labId,a.userId]);
                if(!order.rowCount){await client.query('ROLLBACK');return sendJson(res,404,{error:'not_found'});}
                if(order.rows[0].status!=='uploading'){await client.query('ROLLBACK');return sendJson(res,409,{error:'order_no_longer_uploading'});}
                const file=await client.query('SELECT id,original_state FROM order_files WHERE id=$1 AND order_id=$2 AND lab_id=$3 AND customer_id=$4 FOR UPDATE',[pb1855Exclude[1],orderId,labId,a.userId]);
                if(file.rowCount&&!['pending','deleted'].includes(file.rows[0].original_state)){await client.query('ROLLBACK');return sendJson(res,409,{error:'upload_already_verified'});}
                if(file.rows[0]?.original_state==='pending')await client.query("UPDATE order_files SET original_state='deleted' WHERE id=$1 AND order_id=$2 AND original_state='pending'",[pb1855Exclude[1],orderId]);
                await client.query('COMMIT');
                return sendJson(res,200,{excluded:true,fileId:pb1855Exclude[1]});
            }catch(error){await client.query('ROLLBACK').catch(()=>{});throw error}
            finally{client.release()}
