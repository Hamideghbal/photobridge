function PB1609DiscountEditor({customer,onSaved}){
 const info=pb1585DiscountInfo(customer),[open,setOpen]=x.useState(false),[value,setValue]=x.useState(''),[busy,setBusy]=x.useState(false),[error,setError]=x.useState('');
 const save=async()=>{if(busy||!info)return;const percent=Number(value);if(!value.trim()||!Number.isFinite(percent)||percent<0||percent>100){setError('درصد باید عددی بین صفر تا صد باشد.');return}setBusy(true);setError('');try{await Me('/api/lab/finance/customer-settings',{method:'PUT',body:JSON.stringify({customerId:customer.id,discountEnabled:info.enabled,discountPercent:percent,discountEveryInvoices:info.every})});setOpen(false);try{await onSaved()}catch{setError('تخفیف ذخیره شد؛ تازه‌سازی فهرست انجام نشد.')}}catch{setError('ذخیرهٔ تخفیف انجام نشد؛ دوباره تلاش کنید.')}finally{setBusy(false)}};
 return e.jsxs(e.Fragment,{children:[e.jsx('button',{type:'button',className:'pb1609DiscountEdit',disabled:!info,title:'ویرایش درصد تخفیف','aria-label':'ویرایش درصد تخفیف '+customer.name,onClick:()=>{setValue(String(info.percent));setError('');setOpen(true)},children:e.jsx(PB1585Discount,{customer})}),!open&&error&&e.jsx('small',{role:'alert',children:error}),open&&e.jsx(On,{title:'ویرایش تخفیف · '+customer.name,onClose:()=>{if(!busy)setOpen(false)},children:e.jsxs('div',{className:'pb1609DiscountDialog',children:[e.jsxs('label',{children:['درصد تخفیف',e.jsx('input',{type:'number',min:0,max:100,step:'0.01',inputMode:'decimal',value,onChange:event=>setValue(event.target.value),'aria-label':'درصد تخفیف',autoFocus:true,disabled:busy})]}),e.jsx('p',{children:info.enabled?'اعمال تخفیف طبق تنظیم فعلی: هر '+V(info.every)+' فاکتور.':'تخفیف در تنظیمات فعلی غیرفعال است؛ فقط درصد ذخیره می‌شود.'}),error&&e.jsx('p',{role:'alert',children:error}),e.jsxs('div',{className:'modalActions',children:[e.jsx('button',{type:'button',disabled:busy,onClick:()=>setOpen(false),children:'انصراف'}),e.jsx('button',{type:'button',className:'primary',disabled:busy,onClick:save,children:busy?'در حال ذخیره…':'ذخیره تخفیف'})]})]})})]});
}
