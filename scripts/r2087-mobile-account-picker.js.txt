     for(;;){

       // or the number of verified sessions. Even one verified role is offered.

         dismiss();
         selected=await choose(portals.map(portal=>({portal,name:accounts.some(a=>a.portal===portal)?'حساب فعال · ورود به همین حساب':descriptions[portal]})),false,'',true);
         if(selected==='back'){dismiss();continue}
         if(selected?.startsWith('login:')){
           const portal=selected.slice(6);
           if(!portals.includes(portal)){dismiss();continue}
           const verified=await validSession(portal);
           if(verified?.uncertain){dismiss();message='وضعیت حساب مشخص نشد؛ دوباره بررسی کنید.';continue}
           if(verified){
             try{moving=navigate(portal);if(!moving)dismiss();return}catch{dismiss();message='ورود انجام نشد؛ دوباره انتخاب کنید.';continue}
           }
           // Open the ordinary role-specific sign-in page using the one-use navigation ticket.
           // This never creates a session or clears any other role's cookies.
           needsResume=false;wasHidden=false;blurred=false;
           if(move(routes[portal],true)){moving=true;return}
           dismiss();message='ورود انجام نشد؛ دسترسی ذخیره موقت مرورگر را بررسی کنید.';continue;
         }
       }
       if(!selected)return;
       if(selected==='retry'){dismiss();message='';continue}
       const valid=await validSession(selected);
       if(valid&&!valid.uncertain){
         try{moving=navigate(selected);if(!moving)dismiss();return}catch{dismiss();message='ورود انجام نشد؛ دسترسی ذخیره موقت مرورگر را بررسی و دوباره انتخاب کنید.';continue}
       }
       // No timeout or failed session probe is permission to enter a role.
       dismiss();message='اعتبار این حساب تأیید نشد. دوباره بررسی کنید.';
     }
   }finally{if(!moving){dismiss();if(needsResume&&!authorized){checking=false;resuming=false;pending();scheduleResume()}else ready()}}
 }
