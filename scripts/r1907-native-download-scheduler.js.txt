/* LAB 1.0.104 candidate: independent scan and per-order single-flight.
   This is loaded after the native worker and its existing integrity checks.
   Never change the download manifest, native writer, entitlement or payment guards. */
const pb1907OrderFlights=new Map();
let pb1907TransferActive=0;
const pb1907TransferWaiters=[];
let pb1907RetryDraining=false;
let pb1907SessionEpoch=0;
async function pb1907WithDownloadSlot(action){
 if(pb1907TransferActive>=2)await new Promise(resolve=>pb1907TransferWaiters.push(resolve));
 pb1907TransferActive++;
 try{return await action()}
 finally{pb1907TransferActive--;const next=pb1907TransferWaiters.shift();if(next)next();}
}
function pb1907ScheduleOrder(lab,order,settings){
 const key=String(lab)+':'+String(order?.id||'');
 if(!order?.id)return Promise.resolve();
 const current=pb1907OrderFlights.get(key);
 if(current)return current;
 const epoch=pb1907SessionEpoch;
 let flight;
 flight=Promise.resolve().then(()=>{
  if(epoch!==pb1907SessionEpoch||!window.pbNativeStorage||String(pb1568Work.labId)!==String(lab))return;
  return pb1568ProcessOrder(lab,order,settings);
 }).catch(error=>{
  pb1568Work.lastError=pb1568Problem(error);
 }).finally(()=>{
  if(pb1907OrderFlights.get(key)===flight)pb1907OrderFlights.delete(key);
 });
 pb1907OrderFlights.set(key,flight);
 return flight;
}
/* LAB104 R1908: targeted notification-to-first-byte path.
   Native notification order IDs bypass the expensive accounts/settings/orders LIST scan.
   Server authorization, exact download manifest, disk proof, and device checks remain
   in the existing downloader. Finance never participates in this fast dispatch. */
const pb1908ArrivalFlights=new Map(),pb1908DetailHints=new Map(),pb1908Timing=new Map();
function pb1908Mark(orderId,stage,data){
 const id=String(orderId||'');if(!id)return;
 const entry=pb1908Timing.get(id)||{startAt:Date.now(),stages:{}};
 if(!entry.stages[stage])entry.stages[stage]=Date.now();
 if(data!==undefined)entry[stage+'Info']=String(data).slice(0,180);
 pb1908Timing.set(id,entry);
 while(pb1908Timing.size>60)pb1908Timing.delete(pb1908Timing.keys().next().value);
 if(stage==='first_byte'||stage==='failed'||stage==='complete'){
  console.info('[LAB104 download latency]',JSON.stringify({orderId:id,stage,elapsedMs:Date.now()-entry.startAt,checkpoints:Object.fromEntries(Object.entries(entry.stages).map(([key,t])=>[key,t-entry.startAt])),info:entry[stage+'Info']||null}));
 }
}
function pb1908Progress(id,value){
 if(value?.phase!=='downloading')return;
 const item=pb1908Timing.get(String(id));if(!item)return;
 const bytes=Number(value.loadedBytes||0);
 if(!item.stages.download_preparing)pb1908Mark(id,'download_preparing');
 if(item.initialBytes===undefined)item.initialBytes=bytes;
 // An already verified local file must not be mistaken for a new network byte.
 if(bytes>item.initialBytes&&!item.stages.first_byte)pb1908Mark(id,'first_byte');
}
function pb1908TakeDetail(lab,id){
 const key=String(lab)+':'+String(id),cached=pb1908DetailHints.get(key);
 if(!cached)return null;pb1908DetailHints.delete(key);
 return Date.now()-cached.at<15000&&cached.detail?.order?.id===id&&Array.isArray(cached.detail.files)?cached.detail:null;
}
function pb1908FastArrival(signal,previewDetail){
 if(!window.__PB_LAB104_NATIVE__||!window.pbNativeStorage)return Promise.resolve(false);
 const id=String(signal?.orderId||'');
 const lab=String(signal?.labId||previewDetail?.order?.lab_id||pb1568Work.labId||'');
 if(!/^[0-9a-f-]{36}$/i.test(id)||!lab||pb1568Work.labId&&String(pb1568Work.labId)!==lab)return Promise.resolve(false);
 const key=lab+':'+id,previous=pb1908ArrivalFlights.get(key)||pb1907OrderFlights.get(key);
 if(previous)return previous;
 const startedEpoch=pb1907SessionEpoch;
 const notifiedAt=Number(signal?.notifiedAt)||Date.now();
 if(!pb1908Timing.has(id))pb1908Timing.set(id,{startAt:notifiedAt,stages:{notification:notifiedAt}});
 pb1908Mark(id,'dispatch');
 const flight=(async()=>{
  // Parallel auth (when not yet known), policy and order DETAIL. No listing scan.
  const [account,config,detail]=await Promise.all([
   String(pb1568Work.labId||'')===lab?Promise.resolve({lab:{id:lab}}):Me('/api/auth/me'),
   Me('/api/lab/automation'),
   previewDetail?.order?.id===id&&Array.isArray(previewDetail.files)?Promise.resolve(previewDetail):Me('/api/lab/orders/'+encodeURIComponent(id))
  ]);
  pb1908Mark(id,'detail_and_policy_ready');
  if(startedEpoch!==pb1907SessionEpoch||!window.pbNativeStorage||String(pb1568Work.labId||lab)!==lab)return false;
  if(String(account?.lab?.id||'')!==lab||String(detail?.order?.lab_id||lab)!==lab)return false;
  const order=detail?.order,task=pb1568State(lab,id),settings=config?.settings;
  if(!order||!settings||task?.phase==='complete'||!task&&order.downloaded_at)return false;
  if(!task){
   if(!settings.autoDownloadEnabled||!['sent','received'].includes(String(order.status||''))||Number(order.file_count||0)<1)return false;
   if(order.settlement_mode==='prepaid'&&order.payment_status!=='paid')return false;
   const enabledAt=Number(localStorage.getItem('pb1568:downloadEnabledAt:'+lab))||0,created=Date.parse(order.created_at||'');
   if(enabledAt&&Number.isFinite(created)&&created<enabledAt)return false;
   // A previously uninitialized device must not mass-download historical notifications.
   if(!enabledAt&&(!Number.isFinite(created)||Date.now()-created>120000))return false;
   if(!enabledAt)localStorage.setItem('pb1568:downloadEnabledAt:'+lab,String(Date.now()-120000));
  }
  if(!Array.isArray(detail.files)||!detail.files.length)return false;
  pb1568Work.labId=lab;
  pb1908DetailHints.set(key,{at:Date.now(),detail});
  pb1908Mark(id,'worker_scheduled');
  await pb1907ScheduleOrder(lab,order,settings);
  pb1908Mark(id,'worker_finished');
  return true;
 })().catch(error=>{
  pb1908Mark(id,'failed',String(error?.message||error));
  // The ordinary polling/retry path remains available after a transient fast-path error.
  if(typeof pb1586WakeDownloads==='function')pb1586WakeDownloads();
  return false;
 }).finally(()=>{if(pb1908ArrivalFlights.get(key)===flight)pb1908ArrivalFlights.delete(key);pb1908DetailHints.delete(key)});
 pb1908ArrivalFlights.set(key,flight);
 return flight;
}
