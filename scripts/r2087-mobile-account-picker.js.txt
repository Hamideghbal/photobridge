/* PB2087: choose among independently verified role sessions; no auth credentials are stored. */
 const entryPaths=new Set([
   '/','/photographer','/photographer/login','/lab','/lab/login','/rental','/rental/login'
 ]);
 const normalize=p=>p.replace(/\/+$/,'')||'/';
 const sensitive=()=>{
   const q=new URLSearchParams(location.search);
   return ['logged_out','token','invite','maintenance_preview','reset','code','redirect','walletPayment','pbAdminShop','pbStorePreview','pbDesktopContent','pbEmbeddedStore','impersonation'].some(k=>q.has(k))||
     /(?:impersonation|token|reset)=/i.test(location.hash||'')||
     /\/(?:signup|register|invite|reset|password|recovery)(?:\/|$)/i.test(location.pathname);
 };
 const query=new URLSearchParams(location.search);
 if(query.has('logged_out')){try{localStorage.removeItem(KEY)}catch{};return}
 if(sensitive())return;
 // This flag prevents a second prompt on the full document navigation caused
 // by selecting a role; it is immediately removed and never authorizes access.
 const handoff=query.get('pb2087AccountEntry')==='1';
 if(handoff){
   query.delete('pb2087AccountEntry');
   try{history.replaceState(history.state,'',location.pathname+(query.toString()?'?'+query:'')+(location.hash||''))}catch{}
 }
 const currentPortal=()=>portalFromPath()||(/^\/equipment-rental\/book(?:\/|$)/.test(location.pathname)?'photographer':null);
 const portals=['photographer','lab','rental'];
 const labels={photographer:'عکاسی',lab:'لابراتوار',rental:'فروشگاه'};
 const descriptions={photographer:'ورود به پنل عکاسی',lab:'ورود به پنل مدیریت لابراتوار',rental:'ورود به پنل فروشگاه'};
 let checking=false,chooser=null,wasHidden=false,needsResume=false;
 const pending=()=>document.documentElement.classList.add('pb1493SessionResumePending','pb2087AccountPending');
 const ready=()=>{
   resuming=false;checking=false;
   document.documentElement.classList.remove('pb1493SessionResumePending','pb2087AccountPending');
   window.dispatchEvent(new Event('pb2087-account-ready'));
 };
 const validSession=async portal=>{
   const abort=new AbortController(),timer=setTimeout(()=>abort.abort(),4500);
   try{
     const r=await originalFetch('/api/auth/me',{credentials:'include',cache:'no-store',headers:{'x-photobridge-portal':portal},signal:abort.signal});
     if(!r.ok)return [401,403].includes(r.status)?null:{uncertain:true,portal};
     const data=await r.json();
     if(!roleForIdentity(portal,data)||data?.user?.mustChangePassword||data?.user?.forcePasswordChange)return null;
     const name=portal==='lab'?data.lab?.name:portal==='rental'?data.rentalShop?.name:(data.user?.photographerName||data.user?.studioName||data.user?.name);
     return {portal,name:typeof name==='string'?name.trim().slice(0,100):''};
   }catch{return {uncertain:true,portal}}finally{clearTimeout(timer)}
 };
 function remember(portal){try{localStorage.setItem(KEY,portal)}catch{}}
 function navigate(portal){
   remember(portal);
   const current=currentPortal();
   // Preserve an already open workspace when choosing its current account.
   if(current===portal&&(!/\/login\/?$/.test(location.pathname)))return false;
   location.replace(routes[portal]+'?pb2087AccountEntry=1');
   return true;
 }
 function choose(accounts,uncertain=false,message=''){
   return new Promise(resolve=>{
     const render=()=>{
       if(chooser||!document.body){resolve(null);return}
       const dialog=document.createElement('dialog');
       chooser=dialog;dialog.id='pb2087AccountPicker';dialog.dir='rtl';dialog.setAttribute('aria-labelledby','pb2087AccountTitle');
       const panel=document.createElement('section');panel.className='pb2087AccountPanel';
       const logo=document.createElement('div');logo.className='pb2087AccountBrand';logo.textContent='PhotoBridge';
       const title=document.createElement('h1');title.id='pb2087AccountTitle';title.textContent='انتخاب حساب کاربری';
       const intro=document.createElement('p');intro.textContent='می‌خواهید وارد کدام حساب شوید؟';
       panel.append(logo,title,intro);
       if(message||uncertain){const notice=document.createElement('p');notice.className='pb2090AccountNotice';notice.textContent=message||'بررسی بعضی حساب‌ها کامل نشد. می‌توانید دوباره بررسی کنید.';panel.append(notice);}
       const accountButtons=[];
       let selected=false;
       for(const account of accounts){
         const button=document.createElement('button');button.type='button';button.className='pb2087AccountOption';button.dataset.portal=account.portal;
         const symbol=document.createElement('span');symbol.className='pb2087AccountIcon';symbol.setAttribute('aria-hidden','true');
         symbol.dataset.icon=account.portal; // R2088: role-specific camera, printer and storefront icons.
         const copy=document.createElement('span');copy.className='pb2087AccountCopy';
         const role=document.createElement('strong');role.textContent=labels[account.portal];
         const sub=document.createElement('small');sub.textContent=account.name||descriptions[account.portal];
         copy.append(role,sub);
         const arrow=document.createElement('span');arrow.className='pb2087AccountArrow';arrow.setAttribute('aria-hidden','true');arrow.dir='ltr'; // R2088: CSS draws a fixed left chevron, immune to RTL mirroring.
         button.append(symbol,copy,arrow);
         button.addEventListener('click',()=>{
           if(chooser!==dialog||selected)return;
           selected=true;button.setAttribute('aria-pressed','true');
           for(const option of accountButtons)option.disabled=true;
           // R2090: explicit click is the only path to account entry. Keep the
           // dialog in place during the final server verification.
           resolve(account.portal);
         });
         accountButtons.push(button);panel.append(button);
       }
       if(uncertain||message){const retry=document.createElement('button');retry.type='button';retry.className='pb2090AccountRetry';retry.textContent='بررسی دوباره حساب‌ها';retry.addEventListener('click',()=>{if(selected)return;selected=true;resolve('retry')});panel.append(retry);}
       dialog.append(panel);dialog.addEventListener('cancel',event=>event.preventDefault());
       document.body.append(dialog);
       try{dialog.showModal()}catch{dialog.setAttribute('open','')}
       panel.querySelector('button')?.focus({preventScroll:true});
     };
     if(document.body)render();else document.addEventListener('DOMContentLoaded',render,{once:true});
   });
 }
 const dismiss=()=>{if(chooser){chooser.remove();chooser=null}};
 async function check(cold){
   if(checking||chooser||sensitive()||document.hidden)return;
   checking=true;resuming=true;pending();
   let moving=false,message='';
   try{
     for(;;){
       const results=await Promise.all(portals.map(validSession));
       if(sensitive()||document.hidden){needsResume=true;return}
       const accounts=results.filter(x=>x&&!x.uncertain),uncertain=results.some(x=>x?.uncertain);
       if(!accounts.length&&!uncertain&&!message){
         try{localStorage.removeItem(KEY)}catch{}
         if(normalize(location.pathname)==='/'){moving=true;location.replace('/photographer?pb2087AccountEntry=1')}
         return;
       }
       // R2090: never enter a role based on last preference, response timing,
       // or the number of verified sessions. Even one verified role is offered.
       const selected=await choose(accounts,uncertain,message);
       if(!selected)return;
       if(selected==='retry'){dismiss();message='';continue}
       const valid=await validSession(selected);
       if(valid&&!valid.uncertain){
         moving=navigate(selected);
         if(!moving)dismiss();
         return;
       }
       // No timeout or failed session probe is permission to enter a role.
       dismiss();message='اعتبار این حساب تأیید نشد. دوباره بررسی کنید.';
     }
   }finally{if(!moving){dismiss();ready()}}
 }
 // Mobile OS launches can restore the last workspace rather than start_url.
 // On a fresh top-level workspace navigation, prompt; preserve ordinary
 // same-origin navigation between workspaces after a deliberate selection.
 const internalReferrer=()=>{try{return Boolean(document.referrer)&&new URL(document.referrer).origin===location.origin}catch{return false}};
 const reloaded=()=>{try{return performance.getEntriesByType('navigation')[0]?.type==='reload'}catch{return false}};
 const initial=entryPaths.has(normalize(location.pathname))||Boolean(currentPortal()&&(!internalReferrer()||reloaded()));
 if(!handoff&&initial)void check(true);
 const safeResume=()=>{
   if(!currentPortal()||sensitive())return false;
   const active=document.activeElement;
   if(active?.matches?.('input,textarea,select,[contenteditable="true"]'))return false;
   if(document.querySelector('dialog[open],.modalOverlay,.globalModalLayer,.simplePrintPilot,.uploadChoiceOverlay,.pbR85ScanBack,.pbR100F73ReturnBack,[aria-busy="true"]'))return false;
   return true;
 };
 function resume(){
   if(!needsResume||checking||chooser||document.hidden||!safeResume())return;
   needsResume=false;void check(false);
 }
 document.addEventListener('visibilitychange',()=>{
   if(document.hidden){wasHidden=true;return}
   if(wasHidden){wasHidden=false;needsResume=true}
   resume();
 });
 window.addEventListener('pageshow',event=>{if(event.persisted){needsResume=true;resume()}});
 // Defer the prompt until an active scan/form/modal is finished.
 document.addEventListener('focusout',()=>{if(needsResume)setTimeout(resume,0)});
 document.addEventListener('click',()=>{if(needsResume)setTimeout(resume,0)},true);
