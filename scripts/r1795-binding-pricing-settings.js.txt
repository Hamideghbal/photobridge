function PB1795BindingLabel(name,combined){return combined?'چاپ و صحافی · '+String(name||'صحافی آلبوم'):String(name||'صحافی آلبوم')}
function PB1795BindingPricingSettings({settings:provided,onSaved}){
 const[settings,setSettings]=x.useState(provided||null),[busy,setBusy]=x.useState(false),[error,setError]=x.useState(''),[notice,setNotice]=x.useState('');
 x.useEffect(()=>{if(provided){setSettings(provided);return}let live=true;Me('/api/lab/album-settings').then(r=>{if(live)setSettings(r.settings)}).catch(()=>{if(live)setError('دریافت روش قیمت‌گذاری انجام نشد؛ صفحه را بازخوانی کنید.')});return()=>{live=false}},[provided]);
 const mode=settings?.printTariffCatalog?.albumPricingMode==='combined'?'combined':'separate';
 const change=async value=>{if(busy||value===mode)return;setBusy(true);setError('');setNotice('');try{const result=await Me('/api/lab/print-catalog/type',{method:'PATCH',body:JSON.stringify({albumPricingMode:value})});if(result?.ok!==true||result?.catalog?.albumPricingMode!==value)throw Error('save_unconfirmed');const next={...settings,printTariffCatalog:result.catalog};setSettings(next);onSaved?.(next);setNotice('روش قیمت‌گذاری ذخیره شد. تعرفه صحافی را متناسب با حالت جدید بازبینی کنید.')}catch{setError('ذخیره روش قیمت‌گذاری انجام نشد؛ حالت قبلی حفظ شد.')}finally{setBusy(false)}};
 return e.jsxs('section',{className:'pb1795BindingPricing',children:[e.jsx('h3',{children:'روش قیمت‌گذاری صحافی'}),e.jsxs('label',{children:['محاسبه چاپ و صحافی',e.jsxs('select',{'aria-label':'روش قیمت‌گذاری صحافی',disabled:busy||!settings,value:mode,onChange:event=>change(event.target.value),children:[e.jsx('option',{value:'separate',children:'چاپ و صحافی جداگانه'}),e.jsx('option',{value:'combined',children:'قیمت یکپارچه چاپ و صحافی'})]})]}),e.jsx('p',{children:mode==='combined'?'مبلغ هر لت در ستون «چاپ و صحافی» شامل چاپ است؛ هزینه چاپ جداگانه به فاکتور اضافه نمی‌شود.':'قیمت چاپ و قیمت صحافی جدا وارد می‌شوند و در فاکتور جداگانه محاسبه می‌شوند.'}),e.jsx('small',{children:'تغییر حالت، قیمت‌های واردشده را تبدیل نمی‌کند و فقط برای سفارش‌های جدید اعمال می‌شود. جلد، باکس و خدمات اضافه جدا هستند.'}),notice&&e.jsx('p',{role:'status',children:notice}),error&&e.jsx('p',{role:'alert',children:error})]});
}
