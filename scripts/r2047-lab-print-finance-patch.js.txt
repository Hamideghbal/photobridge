// Pure functions shared by the final server and both Lab UI payloads.
function pb2047CleanUploadName(value){
 const leaf=String(value||'file').replace(/\\/g,'/').split('/').pop()||'file';
 const ext=leaf.match(/\.[A-Za-z0-9]{1,12}$/)?.[0]||'';
 const stem=ext?leaf.slice(0,-ext.length):leaf;
 return (stem.split('=')[0].trim()||'photo')+ext;
}
// If two distinct source filenames collapse to the same cleaned name,
// distinguish a later sequential upload before appending server-owned markers.
// Existing files and idempotent replays are never renamed.
async function pb2047UniqueUploadBase(db,orderId,id,clean){
 const ext=clean.match(/\.[A-Za-z0-9]{1,12}$/)?.[0]||'';
 const base=ext?clean.slice(0,-ext.length):clean;
 const q=await db.query("SELECT original_name FROM order_files WHERE order_id=$1 AND id<>$2 AND (lower(original_name)=lower($3) OR lower(rtrim(split_part(original_name,'=',1)))=lower($4)) LIMIT 1",[orderId,id,clean,base]);
 if(!q.rowCount)return clean;
 return base.slice(0,Math.max(1,120-ext.length))+'-'+String(id).replace(/[^A-Za-z0-9]/g,'').slice(0,12)+ext;
}
function pb2047PrintMarked(order,album){
 if(!order)return false;
 if(album)return ['printing','binding','ready','completed'].includes(String(order.album_stage||''))||!!order.printed_at||['ready','completed','printed'].includes(String(order.status||''));
 return !!order.printed_at||['printing','reviewing','ready_print','printed','ready','completed'].includes(String(order.status||''));
}
function pb2047Downloaded(order){
 return !!order?.downloaded_at||['downloaded','printing','reviewing','ready_print','printed','ready','completed'].includes(String(order?.status||''));
}
function pb2047RangeError(customer,from,to,date){
 if(!customer)return 'لطفاً ابتدا نام عکاسی را انتخاب کنید.';
 if(!from||!to)return 'لطفاً تاریخ شروع و پایان بازه را انتخاب کنید.';
 if(!date(from)||!date(to)||date(from)>date(to))return 'تاریخ شروع نباید بعد از تاریخ پایان باشد.';
 return '';
}
function pb2047InRange(rows,customer,from,to,date){
 return rows.some(row=>String(row.customer_id)===String(customer)&&date(row.issued_at)>=date(from)&&date(row.issued_at)<=date(to));
}
function pb2047One(s,a,b,label){
 const count=s.split(a).length-1;
 if(count!==1)throw Error('[R2047] '+label+' anchor '+count);
 return s.replace(a,()=>b);
}
function pb2047Region(s,a,b,edit){
 const i=s.indexOf(a),j=s.indexOf(b,i+a.length);
 if(i<0||j<=i)throw Error('[R2047] region '+a);
 return s.slice(0,i)+edit(s.slice(i,j))+s.slice(j);
}
