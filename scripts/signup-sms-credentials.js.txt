export function signupSmsIdentity(rows,preferredId){
 const active=rows.filter(r=>r.active===true);
 const user=preferredId?active.find(r=>String(r.id)===String(preferredId)):(active.find(r=>r.platform_role==='platform_admin')||active[0]);
 if(rows.length&&!user)throw fail('existing_account_invalid');
 return user||null;
}
export async function prepareSignupCredential(db,{kind,mobile,payload,password,verifyPassword}){
 if(!['lab_trial','equipment_rental_trial','photographer','invite'].includes(kind))return payload;
 const rows=await credentialRows(db,mobile),user=signupSmsIdentity(rows,payload.existingUserId||payload.existingCredentialUserId);
 if(!user){if(String(password||'').length<8)throw fail('password_too_short',400);return {...payload,sharedCredential:{mobile:sharedMobile(mobile),snapshot:null,revokeSessions:false}};}
 return {...payload,sharedCredential:{mobile:sharedMobile(mobile),snapshot:snapshot(rows),userId:String(user.id),linkOnly:true,revokeSessions:false}};
}
export async function signupCredentialPasswordHash(db,payload,password,hashPassword){
 const proof=payload?.sharedCredential;
 if(!proof?.linkOnly)return hashPassword(password);
 const rows=await credentialRows(db,proof.mobile),user=signupSmsIdentity(rows,proof.userId);
 if(snapshot(rows)!==proof.snapshot)throw fail('existing_account_invalid');
 const hash=user.password_hash||user.photographer_password_hash;
 if(!/^[0-9a-f]{32}:[0-9a-f]{128}$/i.test(String(hash||'')))throw fail('account_password_unification_required');
 return hash;
}
export async function finishSignupCredential(db,userId,challenge){
 const proof=challenge.payload?.sharedCredential;if(!proof)return;
 if(!proof.linkOnly)return setSharedPassword(db,userId,String(challenge.password_hash),{proof:true,mobile:challenge.mobile,revokeSessions:proof.revokeSessions!==false});
 if(String(userId)!==proof.userId||sharedMobile(challenge.mobile)!==proof.mobile)throw fail('existing_account_invalid');
 const verified=await db.query("SELECT payload FROM signup_mobile_verifications WHERE id=$1 AND status='verified' AND mobile=$2",[challenge.id,challenge.mobile]);
 if(!verified.rowCount)throw fail('verification_required');
 const stored=verified.rows[0].payload?.sharedCredential;
 if(!stored?.linkOnly||stored.userId!==proof.userId||stored.snapshot!==proof.snapshot)throw fail('existing_account_invalid');
 // The SMS proves ownership; adding a role must never change existing credentials.
 await db.query('UPDATE users SET mobile_verified_at=COALESCE(mobile_verified_at,now()) WHERE id=$1',[userId]);
}
