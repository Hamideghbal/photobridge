// The category/section buttons remain available and the user can retry safely.
class PB1540WalkinBoundary extends x.Component {

 render(){return this.state.failed?e.jsxs('section',{className:'card panel pb1540WalkinFallback',children:[e.jsx('h3',{children:'بخش مشتری آزاد موقتاً باز نشد'}),e.jsx('p',{children:'می‌توانید از بخش‌های دیگر استفاده کنید یا دریافت اطلاعات مشتری آزاد را دوباره امتحان کنید.'}),e.jsx('button',{type:'button',onClick:()=>this.setState({failed:false}),children:'تلاش دوباره'})]}):this.props.children}
}
