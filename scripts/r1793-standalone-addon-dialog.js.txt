/* R2040: each model has its own quantity, price, stock and frame colour. */
function PB2040AddonLines(row,models,quantities,colorIds,priceFor,stockFor){
 const lines=[];let total=0,count=0,error="";
 if(!row)return {lines,total,count,error:"ابتدا سایز را انتخاب کنید."};
 for(const model of models){
  const raw=quantities[model.id];
  if(raw==null||raw===""||Number(raw)===0)continue;
  const qty=Number(raw),unit=priceFor(model);
  if(!Number.isSafeInteger(qty)||qty<1||qty>99){error="تعداد "+model.name+" باید بین ۰ تا ۹۹ باشد.";break}
  if(unit==null||!Number.isSafeInteger(unit)||unit<0){error="تعرفه "+model.name+" موجود نیست.";break}
  const stock=stockFor(model);
  if(stock!=null&&stock!==""&&Number(stock)<qty){error="موجودی "+model.name+" برای تعداد انتخاب‌شده کافی نیست.";break}
  const colors=model.kind==="frame"?(model.colors||[]).filter(c=>c.available!==false):[];
  const color=colors.find(c=>c.id===colorIds[model.id])||(colors.length===1||model.legacy?colors[0]:null)||null;
  if(colors.length&&!color){error="رنگ "+model.name+" را انتخاب کنید.";break}
  lines.push({row,chassis:model.kind==="chassis"?model:null,frame:model.kind==="frame"?model:null,color,quantity:qty});
  total+=unit*qty;count+=qty;
 }
 if(!error&&!lines.length)error="حداقل برای یک مدل تعداد وارد کنید.";
 return {lines,total,count,error};
}
function PB1793AddonOnlyDialog({prices=[],catalog={},onClose,onAdd}){
 const [sizeId,setSizeId]=x.useState(""),[quantities,setQuantities]=x.useState({}),[colorIds,setColorIds]=x.useState({});
 const [busy,setBusy]=x.useState(false),[error,setError]=x.useState("");
 const row=prices.find(p=>String(p.id)===String(sizeId));
 const all=Array.isArray(catalog.types)?catalog.types:[];
 const priceFor=model=>{
  if(!row||!model)return null;
  const raw=model.legacy||model.id===model.kind?row[model.kind]:catalog.rates?.[row.id]?.[model.id];
  return raw==null||raw===""||!Number.isSafeInteger(Number(raw))||Number(raw)<0?null:Number(raw);
 };
 const stockFor=model=>!model?null:model.legacy||model.id===model.kind?row?.[model.kind+"_stock"]:catalog.stocks?.[row.id]?.[model.id];
 const models=kind=>all.filter(model=>model.kind===kind&&model.active!==false&&priceFor(model)!=null);
 const available=[...models("chassis"),...models("frame")];
 const draft=PB2040AddonLines(row,available,quantities,colorIds,priceFor,stockFor);
 const lock=x.useRef(false);
 const save=async()=>{
  if(lock.current)return;
  if(draft.error){setError(draft.error);return}
  lock.current=true;setBusy(true);setError("");
  try{await onAdd(draft.lines)}
  catch(err){setError(String(err?.message||"ثبت سفارش بدون چاپ انجام نشد."))}
  finally{lock.current=false;setBusy(false)}
 };
 const renderModel=m=>{
  const stock=stockFor(m),qty=quantities[m.id]??"",colors=m.kind==="frame"?(m.colors||[]).filter(c=>c.available!==false):[];
  const selectedColor=colors.find(c=>c.id===colorIds[m.id])||(colors.length===1||m.legacy?colors[0]:null);
  const n=Number(qty),valid=Number.isSafeInteger(n)&&n>=0&&n<=99;
  const id="pb2040-addon-"+String(m.id);
  return e.jsxs("div",{className:"pb2040AddonModel",style:{display:"grid",gridTemplateColumns:"minmax(0,1fr) 76px",gap:"7px 12px",alignItems:"center",padding:"9px 0",borderBottom:"1px solid #dce5f0"},children:[
   e.jsxs("label",{htmlFor:id,children:[e.jsx("strong",{children:m.name}),e.jsxs("small",{children:[V(priceFor(m))," تومان / عدد",stock!=null&&stock!==""?" · موجودی "+V(Number(stock)):""]})]}),
   e.jsx("input",{id,type:"number",min:0,max:99,step:1,inputMode:"numeric",placeholder:"۰",disabled:busy||Number(stock)===0&&stock!=null&&stock!=="",value:qty,"aria-label":"تعداد "+m.name,onChange:ev=>{const value=ev.target.value;setQuantities(prev=>({...prev,[m.id]:value}));setError("")}}),
   !!colors.length&&n>0&&e.jsxs("label",{style:{gridColumn:"1/-1"},children:["رنگ "+m.name,e.jsxs("select",{disabled:busy,value:selectedColor?.id||"",onChange:ev=>{const value=ev.target.value;setColorIds(prev=>({...prev,[m.id]:value}));setError("")},children:[e.jsx("option",{value:"",children:"انتخاب رنگ قاب"}),...colors.map(c=>e.jsx("option",{value:c.id,children:c.name||"رنگ قاب"},c.id))]})]}),
   e.jsx("small",{style:{gridColumn:"1/-1",textAlign:"end"},children:"مبلغ: "+V(valid?priceFor(m)*n:0)+" تومان"})
  ]},m.id);
 };
 return e.jsx(On,{title:"سفارش شاسی و قاب بدون چاپ",onClose:busy?()=>{}:onClose,children:e.jsxs("div",{className:"pb1793AddonDialog pb2040AddonDialog",children:[
  e.jsx("p",{className:"pb1793Hint",children:"سایز را انتخاب کنید و تعداد هر مدل را جداگانه وارد کنید. صفر یا خالی یعنی عدم سفارش آن مدل."}),
  e.jsxs("label",{children:["سایز محصول",e.jsxs("select",{disabled:busy,value:sizeId,onChange:ev=>{setSizeId(ev.target.value);setQuantities({});setColorIds({});setError("")},children:[e.jsx("option",{value:"",children:"انتخاب سایز"}),...prices.map(p=>e.jsx("option",{value:String(p.id),children:String(p.label||p.id)},p.id))]})]}),
  !!row&&["chassis","frame"].map(kind=>e.jsxs("section",{children:[e.jsx("strong",{children:kind==="chassis"?"مدل‌های شاسی":"مدل‌های قاب"}),models(kind).length?models(kind).map(renderModel):e.jsx("p",{className:"pb1793Hint",children:"مدلی با تعرفه فعال برای این سایز موجود نیست."})]},kind)),
  !!row&&e.jsxs("div",{className:"pb1793Total",children:[e.jsxs("span",{children:["تعداد کل: ",V(draft.count)]}),e.jsx("strong",{children:V(draft.total)+" تومان"})]}),
  (error||row&&available.some(m=>Number(quantities[m.id])>0)&&draft.error)&&e.jsx("p",{role:"alert",className:"pb1793Error",children:error||draft.error}),
  e.jsxs("div",{className:"pb1793Actions",children:[e.jsx("button",{type:"button",disabled:busy,onClick:onClose,children:"انصراف"}),e.jsx("button",{type:"button",className:"primary",disabled:busy||!!draft.error,onClick:save,children:busy?"در حال افزودن…":"افزودن به سفارش"})]})
 ]})});
}
