function PB1543OpeningDebtPreview({customerId,amount,updatedAt}){
 const ref=x.useRef(null),[preview,setPreview]=x.useState(null),[failed,setFailed]=x.useState(false);
 x.useEffect(()=>{if(!customerId||amount!==undefined)return;let alive=true,started=false;const controller=new AbortController();const read=async()=>{if(started)return;started=true;try{const result=await Me(pb1544DebtUrl(customerId,'account-summary'),{signal:controller.signal});if(alive){if(!Number.isFinite(Number(result.initialDebt)))throw Error('missing_amount');setPreview(result)}}catch{if(alive)setFailed(true)}};let observer;if(typeof IntersectionObserver==='function'&&ref.current){observer=new IntersectionObserver(entries=>{if(entries.some(e=>e.isIntersecting)){observer.disconnect();void read()}},{rootMargin:'120px'});observer.observe(ref.current)}else void read();return()=>{alive=false;controller.abort();observer?.disconnect()}},[customerId,amount]);
 const value=amount!==undefined?Number(amount):preview?.initialDebt,date=amount!==undefined?updatedAt:preview?.initialDebtUpdatedAt;
 return e.jsxs('small',{ref,className:'pb1543OpeningDebtAmount',role:'status',children:[value!=null&&Number.isFinite(Number(value))?'بدهی اولیه: '+Number(value).toLocaleString('fa-IR')+' تومان':failed?'مبلغ بدهی در دسترس نیست':'دریافت مبلغ بدهی…',date&&e.jsx('time',{dateTime:date,children:pb1586DebtDate(date)})]});
